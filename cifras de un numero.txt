        Scanner ent = new Scanner (System.in);
        int num;
        
        System.out.println("Ingresa un numero: ");
        num=ent.nextInt();
        
        if(num<10){
            System.out.println("Tu numero tiene una cifra");
        }else if(num<100){

        }     
            
        
    }
